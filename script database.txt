DROP DATABASE IF EXISTS TecnoStoreOscar_DB;

CREATE DATABASE TecnoStoreOscar_DB;

USE TecnoStoreOscar_DB;



CREATE TABLE roles (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(30) NOT NULL UNIQUE
);


CREATE TABLE usuarios (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    usuario VARCHAR(50) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,
    id_rol INT NOT NULL,

    CONSTRAINT fk_usuario_rol
        FOREIGN KEY (id_rol)
        REFERENCES roles(id)
        ON UPDATE CASCADE
        ON DELETE RESTRICT
);


CREATE TABLE marcas (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(50) NOT NULL UNIQUE
);


CREATE TABLE sistemas_operativos (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(50) NOT NULL UNIQUE
);


CREATE TABLE gamas (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(30) NOT NULL UNIQUE
);


CREATE TABLE celulares (
    id INT AUTO_INCREMENT PRIMARY KEY,

    marca_id INT NOT NULL,
    modelo VARCHAR(100) NOT NULL,
    sistema_operativo_id INT NOT NULL,
    gama_id INT NOT NULL,

    precio DECIMAL(12,2) NOT NULL,
    stock INT NOT NULL DEFAULT 0,

    CONSTRAINT fk_celular_marca
        FOREIGN KEY (marca_id)
        REFERENCES marcas(id)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,

    CONSTRAINT fk_celular_so
        FOREIGN KEY (sistema_operativo_id)
        REFERENCES sistemas_operativos(id)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,

    CONSTRAINT fk_celular_gama
        FOREIGN KEY (gama_id)
        REFERENCES gamas(id)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,

    CONSTRAINT chk_celular_precio
        CHECK (precio > 0),

    CONSTRAINT chk_celular_stock
        CHECK (stock >= 0)
);



CREATE TABLE clientes (
    id INT AUTO_INCREMENT PRIMARY KEY,

    nombre VARCHAR(100) NOT NULL,
    identificacion VARCHAR(30) NOT NULL UNIQUE,
    correo VARCHAR(150) NOT NULL UNIQUE,
    telefono VARCHAR(20) NOT NULL
);



CREATE TABLE ventas (
    id INT AUTO_INCREMENT PRIMARY KEY,

    cliente_id INT NOT NULL,
    vendedor_id INT NOT NULL,

    fecha DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,

    subtotal DECIMAL(12,2) NOT NULL,
    iva DECIMAL(12,2) NOT NULL,
    total DECIMAL(12,2) NOT NULL,

    CONSTRAINT fk_venta_cliente
        FOREIGN KEY (cliente_id)
        REFERENCES clientes(id)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,

    CONSTRAINT fk_venta_vendedor
        FOREIGN KEY (vendedor_id)
        REFERENCES usuarios(id)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,

    CONSTRAINT chk_venta_subtotal
        CHECK (subtotal >= 0),

    CONSTRAINT chk_venta_iva
        CHECK (iva >= 0),

    CONSTRAINT chk_venta_total
        CHECK (total >= 0)
);


CREATE TABLE detalle_ventas (
    id INT AUTO_INCREMENT PRIMARY KEY,

    venta_id INT NOT NULL,
    celular_id INT NOT NULL,

    cantidad INT NOT NULL,
    precio_unitario DECIMAL(12,2) NOT NULL,
    subtotal DECIMAL(12,2) NOT NULL,

    CONSTRAINT fk_detalle_venta
        FOREIGN KEY (venta_id)
        REFERENCES ventas(id)
        ON UPDATE CASCADE
        ON DELETE CASCADE,

    CONSTRAINT fk_detalle_celular
        FOREIGN KEY (celular_id)
        REFERENCES celulares(id)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,

    CONSTRAINT chk_detalle_cantidad
        CHECK (cantidad > 0),

    CONSTRAINT chk_detalle_precio
        CHECK (precio_unitario > 0),

    CONSTRAINT chk_detalle_subtotal
        CHECK (subtotal >= 0)
);
